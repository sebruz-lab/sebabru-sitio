# Las paginas que no deben aparecer en Google llevan <meta name="robots"
# content="noindex"> en su propio <head>, no un Disallow aca. Un Disallow
# impide rastrearlas, y entonces Google nunca lee el noindex: si alguna ya
# estaba indexada, se queda indexada para siempre.

# /textos/ son los 36 textos de la calculadora en JSON, para que los lea
# /cartas (ver interpretaciones.js en el repo tucarta). El mismo texto, para
# leer, esta en /sol/, /lunas/ y /ascendente/: esas son las paginas que tienen
# que estar en Google, no el JSON. Aca el Disallow no tiene el problema del
# noindex, porque no son paginas que puedan quedar indexadas.
Disallow: /textos/
